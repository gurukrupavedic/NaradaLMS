const CMChapterEditorView = ({ chapterId, trackId, tracks, chaptersData, chapterContentData, onBackToTrack, onSaveChapterContent, onPublishChapter }) => {
  const [activeTab, setActiveTab] = useState('text_content');
  const [currentChapterData, setCurrentChapterData] = useState(null); 
  const [activeTextLang, setActiveTextLang] = useState(CM_SCRIPT_OPTIONS[0].id);
 
  const isPublished = useMemo(() => currentChapterData?.status === 'published', [currentChapterData]);

  useEffect(() => {
    const chapterDetailsList = chaptersData[trackId] || [];
    const chapterMetadata = chapterDetailsList.find(ch => ch.id === chapterId);
   
    if (!chapterMetadata && !chapterContentData[chapterId]) {
        console.error(`ChapterEditorView: Chapter with ID ${chapterId} not found in track ${trackId} or in content details. Cannot load editor.`);
        setCurrentChapterData(null); 
        return;
    }

    const content = chapterContentData[chapterId] || 
                      { 
                        title: chapterMetadata?.title || 'New Chapter', 
                        order: chapterMetadata?.order || 1, 
                        texts: { te: '', hi: '', en: '' }, 
                        audioFiles: [], 
                        segments: [], 
                        mappings: {}, 
                        status: 'draft' 
                      };
    setCurrentChapterData({...content, trackId: trackId}); 
    if (!content.texts) { 
        setCurrentChapterData(prev => ({...prev, texts: { te: '', hi: '', en: '' }}));
    }
  }, [chapterId, trackId, chaptersData, chapterContentData]);

  const handleTextChange = (lang, value) => {
    setCurrentChapterData(prev => ({
      ...prev,
      texts: {
        ...(prev?.texts || {}), 
        [lang]: value
      }
    }));
  };
 
  const handleTitleChange = (e) => {
    setCurrentChapterData(prev => ({...prev, title: e.target.value }));
  };
  const handleOrderChange = (e) => {
     setCurrentChapterData(prev => ({...prev, order: parseInt(e.target.value, 10) || 1 }));
  };

  const handleSaveDraft = () => {
    onSaveChapterContent(chapterId, currentChapterData, trackId); 
    alert(`Chapter "${currentChapterData?.title}" draft saved (mock).`); // User's original code uses alert
  };

  const handleTogglePublish = () => {
    const newStatus = isPublished ? 'draft' : 'published';
    const updatedChapter = { ...currentChapterData, status: newStatus };
    setCurrentChapterData(updatedChapter); 
    onPublishChapter(chapterId, updatedChapter, trackId); 
    alert(`Chapter "${currentChapterData?.title}" status changed to ${newStatus}. Editing is now ${newStatus === 'published' ? 'disabled' : 'enabled'}.`); // User's original code uses alert
  };
 
  const track = tracks.find(t => t.id === trackId);

  if (!currentChapterData) {
    return (
      <div className="p-6 text-center">
        <AlertTriangle size={48} className="mx-auto text-red-500 mb-4" />
        <p className="text-xl font-semibold text-red-700">Error: Chapter Data Not Found</p>
        <p className="text-gray-600">The details for this chapter could not be loaded.</p>
        <button onClick={() => onBackToTrack(trackId)} className="mt-6 flex items-center justify-center mx-auto gap-2 px-4 py-2 bg-blue-600 text-white rounded-md hover:bg-blue-700 text-sm">
            <ChevronLeft size={18} /> Go Back to Chapters in "{track?.title || 'Track'}"
        </button>
      </div>
    );
  }

  const chapterTabs = [
    { id: 'text_content', label: 'Text Content', icon: FileText },
    { id: 'media', label: 'Media', icon: Music },
    { id: 'segmentation_mapping', label: 'Segmentation & Mapping', icon: Link2 },
    { id: 'preview', label: 'Preview', icon: Eye },
  ];

  return (
    <div className="p-4 sm:p-6">
      <button onClick={() => onBackToTrack(trackId)} className="flex items-center gap-2 text-blue-600 hover:text-blue-800 mb-4 text-sm">
        <ChevronLeft size={18} /> Back to Chapters in "{track?.title}"
      </button>

const ContentManagementPortalView = (props) => { // Wrapper for CM views
  const [currentCMView, setCurrentCMView] = useState('dashboard'); 
  const [selectedCMTrackId, setSelectedCMTrackId] = useState(null);
  const [selectedCMChapterId, setSelectedCMChapterId] = useState(null);

  if (currentCMView === 'dashboard') {
    return <CMContentDashboardView 
              tracks={props.contentTracksData} 
              onCreateTrack={props.onCreateTrack} 
              onSelectTrack={(id) => { setSelectedCMTrackId(id); setCurrentCMView('track_editor');}} 
              onDeleteTrack={props.onDeleteTrack} 
            />;
  }
  if (currentCMView === 'track_editor' && selectedCMTrackId) {
    const track = props.contentTracksData.find(t => t.id === selectedCMTrackId);
    const chapters = props.contentChaptersByTrackData[selectedCMTrackId] || [];
    if (!track) { 
        console.warn(`TrackEditorView: Track with ID ${selectedCMTrackId} not found. Returning to dashboard.`);
        setCurrentCMView('dashboard');
        setSelectedCMTrackId(null);
        return <CMContentDashboardView {...props} />; 
    }
    return <CMTrackEditorView 
              track={track} 
              chapters={chapters} 
              onBack={() => {setCurrentCMView('dashboard'); setSelectedCMTrackId(null); setSelectedCMChapterId(null);}}
              onSaveTrack={props.onSaveTrack}
              onCreateChapter={props.onCreateChapter}
              onSelectChapter={(id) => {setSelectedCMChapterId(id); setCurrentCMView('chapter_editor');}}
              onDeleteChapter={props.onDeleteChapter}
            />;
  }
  if (currentCMView === 'chapter_editor' && selectedCMChapterId && selectedCMTrackId) {
    return <CMChapterEditorView 
              chapterId={selectedCMChapterId} 
              trackId={selectedCMTrackId}
              tracks={props.contentTracksData}
              chaptersData={props.contentChaptersByTrackData}
              chapterContentData={props.contentChapterDetailsData}
              onBackToTrack={(tid) => {setCurrentCMView('track_editor'); setSelectedCMChapterId(null); /* Keep selectedCMTrackId */}}
              onSaveChapterContent={props.onSaveChapterContent}
              onPublishChapter={props.onPublishChapter}
            />;
  }
  // Fallback to dashboard if state is inconsistent
  return <CMContentDashboardView {...props} />;
};

    setContentChaptersByTrackData(prev => { 
        const newChapters = {...prev};
        delete newChapters[trackIdToDelete];
        return newChapters;
    });
    setContentChapterDetailsData(prevContent => {
        const newContent = {...prevContent};
        chapterIdsToDelete.forEach(chId => delete newContent[chId]);
        return newContent;
    });
    alert(`Content Track (ID: ${trackIdToDelete}) and its chapters deleted.`); // User's original code uses alert
  };
  const handleCMSaveTrack = (updatedTrackData) => {
     setContentTracksData(prev => prev.map(t => t.id === updatedTrackData.id ? {...t, ...updatedTrackData, lastModified: new Date().toISOString().split('T')[0]} : t));
     alert(`Content Track "${updatedTrackData.title}" saved.`); // User's original code uses alert
  };
  const handleCMCreateChapter = (trackIdForChapter, newChapterData) => {
    const newChapter = { ...newChapterData, id: `cm_ch_${trackIdForChapter}_${Date.now()}`, trackId: trackIdForChapter, status: 'draft', completeness: {text: false, audio: false, segments: false, mapping: false } };
    setContentChaptersByTrackData(prev => ({
        ...prev,
        [trackIdForChapter]: [...(prev[trackIdForChapter] || []), newChapter]
    }));
    setContentChapterDetailsData(prevContent => ({ // Initialize details for the new chapter
        ...prevContent,
        [newChapter.id]: {
            title: newChapter.title, order: newChapter.order, texts: { te: '', hi: '', en: '' },
            audioFiles: [], segments: [], mappings: {}, status: 'draft', trackId: trackIdForChapter
        }
    }));
    setContentTracksData(prevTracks => prevTracks.map(t => t.id === trackIdForChapter ? {...t, chapterCount: (t.chapterCount || 0) + 1} : t));
    alert(`Chapter "${newChapter.title}" created in track "${contentTracksData.find(t=>t.id === trackIdForChapter)?.title}".`); // User's original code uses alert
  };
  const handleCMDeleteChapter = (trackIdOfChapter, chapterIdToDelete) => {
    setContentChaptersByTrackData(prev => ({
        ...prev,
        [trackIdOfChapter]: (prev[trackIdOfChapter] || []).filter(ch => ch.id !== chapterIdToDelete)
    }));
    setContentChapterDetailsData(prevContent => {
        const newContent = {...prevContent};
        delete newContent[chapterIdToDelete];
        return newContent;
    });
    setContentTracksData(prevTracks => prevTracks.map(t => t.id === trackIdOfChapter ? {...t, chapterCount: Math.max(0, (t.chapterCount || 0) - 1)} : t));
    alert(`Chapter (ID: ${chapterIdToDelete}) deleted from track "${contentTracksData.find(t=>t.id === trackIdOfChapter)?.title}".`); // User's original code uses alert
  };
  // This handler updates the detailed content AND the summary in contentChaptersByTrackData
  const handleCMSaveChapterContent = (chapterIdToSave, updatedData, trackIdOfChapter) => {
    // Update detailed content
    setContentChapterDetailsData(prev => ({...prev, [chapterIdToSave]: {...prev[chapterIdToSave], ...updatedData }}));
   
    // Update the summary data in contentChaptersByTrackData (title, order, status, completeness)
    setContentChaptersByTrackData(prevChs => {
        const newChaptersForTrack = (prevChs[trackIdOfChapter] || []).map(ch => 
            ch.id === chapterIdToSave ? { // Update the specific chapter
                ...ch, 
                title: updatedData.title, 
                order: updatedData.order, 
                status: updatedData.status, // Ensure status is also updated in the summary list
                completeness: { // Recalculate completeness based on the detailed data
                    text: Object.values(updatedData.texts || {}).some(t => t && t.trim().length > 5), // More robust check
                    audio: (updatedData.audioFiles?.length || 0) > 0,
                    segments: (updatedData.segments?.length || 0) > 0,
                    mapping: Object.values(updatedData.mappings || {}).some(m => Object.keys(m).length > 0)
                }
            } : ch
        );
        return {...prevChs, [trackIdOfChapter]: newChaptersForTrack };
    });
  };
  const handleCMPublishChapter = (chapterIdToPublish, updatedData, trackIdOfChapter) => {
    // Publishing also implies saving, so call the save content handler
    handleCMSaveChapterContent(chapterIdToPublish, updatedData, trackIdOfChapter);
    // In a real app, this might trigger additional backend processes for publishing.
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-orange-50 via-amber-50 to-yellow-50 font-sans text-slate-800">
      {currentView === 'unifiedDashboard' && (
        <UnifiedDashboardView 
            learningTracks={LEARNING_TRACKS_DATA} 
            instructorStudents={instructorStudents} 
            setCurrentView={setCurrentView} 
            setSelectedTrack={setSelectedTrack} 
            setSelectedStudentForManagement={setSelectedStudentForManagement} 
            studentProficiency={currentUserProficiency} 
            getLevelColor={memoizedGetLevelColor} 
            getLevelLabel={memoizedGetLevelLabel} 
            isTrackNotStarted={memoizedIsStudentTrackNotStarted} 
            isTrackComplete={memoizedIsStudentTrackComplete} 
            isTrackReadyForTest={memoizedIsStudentTrackReadyForTest}
            currentAppUserRoles={currentAppUserRoles}
            // Admin Props
            adminUsers={adminUsers}
            onSaveAdminUser={handleSaveAdminUser}
            onResendInvite={handleResendAdminInvite} // Prop name matches UnifiedDashboardView
            onToggleUserStatus={handleToggleAdminUserStatus}
            // Content Manager Props
            contentTracksData={contentTracksData}
            contentChaptersByTrackData={contentChaptersByTrackData}
            contentChapterDetailsData={contentChapterDetailsData}
            onCreateTrack={handleCMCreateTrack}
            onDeleteTrack={handleCMDeleteTrack}
            onSaveTrack={handleCMSaveTrack}
            onCreateChapter={handleCMCreateChapter}
            onDeleteChapter={handleCMDeleteChapter}
            onSaveChapterContent={handleCMSaveChapterContent}
            onPublishChapter={handleCMPublishChapter}
        /> 
      )}
      {currentView === 'studentTrackView' && selectedTrack && ( <StudentTrackView selectedTrack={selectedTrack} setCurrentView={setCurrentView} setSelectedChapter={setSelectedChapter} studentProficiency={currentUserProficiency} getLevelColor={memoizedGetLevelColor} getLevelLabel={memoizedGetLevelLabel} isTrackComplete={memoizedIsStudentTrackComplete} isTrackReadyForTest={memoizedIsStudentTrackReadyForTest} /> )}
      {currentView === 'studentChapterView' && selectedChapter && selectedTrack && ( <StudentChapterView selectedChapter={selectedChapter} selectedTrack={selectedTrack} setCurrentView={setCurrentView} audioRef={audioRef} studentProficiency={currentUserProficiency} getLevelColor={memoizedGetLevelColor} getLevelLabel={memoizedGetLevelLabel}/> )}
      {currentView === 'instructorStudentProficiencyManagementView' && selectedStudentForManagement && ( <StudentProficiencyManagementView student={selectedStudentForManagement} tracks={LEARNING_TRACKS_DATA} studentProficiencies={studentProficiencies} updateStudentProficiency={updateStudentProficiency} setCurrentView={setCurrentView} setSelectedStudentForManagement={setSelectedStudentForManagement} getLevelColor={memoizedGetLevelColor} getLevelLabel={memoizedGetLevelLabel}/> )}
      <audio ref={audioRef} className="hidden" />
    </div>
  );
};
export default App;
