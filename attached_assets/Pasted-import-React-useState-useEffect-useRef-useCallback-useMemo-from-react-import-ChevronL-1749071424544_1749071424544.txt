// Helper function to parse HH:MM:SS or MM:SS or SS to seconds
const parseTimestampToSeconds = (ts) => {
  if (!ts || typeof ts !== 'string') return 0;
  const parts = ts.split(':').map(Number);
  if (parts.length === 3) { // HH:MM:SS
    return parts[0] * 3600 + parts[1] * 60 + parts[2];
  }
  if (parts.length === 2) { // MM:SS
    return parts[0] * 60 + parts[1];
  }
  if (parts.length === 1) { // SS
    return parts[0];
  }
  return 0;
};

// Audio Time Formatter
const formatTime = (timeInSeconds) => {
  const seconds = Math.floor(timeInSeconds % 60);
  const minutes = Math.floor((timeInSeconds / 60) % 60);
  const hours = Math.floor(timeInSeconds / 3600);
  if (hours > 0) {
    return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
  }
  return `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
};

// Segment Renderer Component
const SegmentRenderer = ({ segments, currentPlayingSegmentId, hoveredSegmentId, onSegmentClick, onSegmentMouseEnter, onSegmentMouseLeave, segmentsContainerRef, selectedScript }) => {
  // Determine font class based on selected script
  const getFontClass = (script) => {
    if (script === 'te') return 'font-tiro-telugu';
    if (script === 'hi' || script === 'en') return 'font-tiro-devanagari-sanskrit'; // Assuming Devanagari font can handle IAST well
    return 'font-mono'; // Fallback
  };

  return (
    <div
      ref={segmentsContainerRef}
      className={`text-xl md:text-2xl leading-loose md:leading-loose text-gray-800 bg-yellow-50 p-4 sm:p-6 rounded-lg border border-yellow-200 ${getFontClass(selectedScript)}`}
    >
      {segments.reduce((acc, segment, index) => {
        acc.push(
          <span
            key={segment.id}
            data-segment-id={segment.id}
            onClick={() => onSegmentClick(segment)}
            onMouseEnter={() => onSegmentMouseEnter(segment.id)}
            onMouseLeave={onSegmentMouseLeave}
            tabIndex={0} // Make it focusable
            onKeyPress={(e) => { if (e.key === 'Enter' || e.key === ' ') onSegmentClick(segment); }} // Keyboard accessible
            className={`cursor-pointer p-0.5 rounded transition-colors duration-150 ease-in-out hover:bg-yellow-200 focus:bg-yellow-300 focus:outline-none ${
              currentPlayingSegmentId === segment.id ? 'bg-yellow-300 font-semibold text-indigo-700' : ''
            } ${
              hoveredSegmentId === segment.id && currentPlayingSegmentId !== segment.id ? 'bg-yellow-200' : ''
            }`}
          >
            {segment.texts[selectedScript] || segment.texts.en /* Fallback to English if script text not found */ }
          </span>
        );
        // Add a non-breaking space if it's not the last segment and no line break is specified
        if (!segment.lineBreakAfter && index < segments.length - 1) {
          acc.push(<span key={`sep-${segment.id}`} className="text-gray-400 mx-0.5 select-none" aria-hidden="true"> </span>);
        }
        // Add a line break if specified
        if (segment.lineBreakAfter) {
          acc.push(<br key={`br-${segment.id}`} />);
        }
        return acc;

const StudentChapterView = ({ selectedChapter, selectedTrack, setCurrentView, audioRef, studentProficiency, getLevelColor, getLevelLabel }) => {
  const chapterProficiencyLevel = studentProficiency[`${selectedTrack.id}_${selectedChapter.id}`] ?? 0;
  if (!selectedChapter || !selectedTrack) return null; // Ensure both are present

  const playerControlsRef = useRef(null); // Ref for the player controls container
  const segmentsContainerRef = useRef(null); // Ref for the segments container

  const chapterKey = `${selectedChapter.trackId}_${selectedChapter.id}`;
  const currentInteractiveData = INTERACTIVE_CHAPTER_DATA[chapterKey];
  const currentChapterFullTextData = STATIC_CHAPTER_DATA[chapterKey];

  // Check if actual interactive segments exist (not just the placeholder)
  const hasActualInteractiveSegments = currentInteractiveData && currentInteractiveData.segments && currentInteractiveData.segments.length > 0 && !currentInteractiveData.segments[0].id.startsWith('ph_seg_');

  // Audio State
  const [selectedAudioSourceId, setSelectedAudioSourceId] = useState(currentInteractiveData?.audioSources[0]?.id || null);
  const [currentPlayingSegmentId, setCurrentPlayingSegmentId] = useState(null);
  const [hoveredSegmentId, setHoveredSegmentId] = useState(null);
  const [isPlaying, setIsPlaying] = useState(false);
  const [playbackSpeed, setPlaybackSpeed] = useState(1);
  const [audioCurrentTime, setAudioCurrentTime] = useState(0);
  const [audioDuration, setAudioDuration] = useState(0);
  const [toaster, setToaster] = useState({ show: false, message: '', type: 'info' }); // Toaster state
  const [selectedScript, setSelectedScript] = useState('te'); // Default to Telugu

  // Toaster display function
  const displayToaster = useCallback((message, type = 'info', duration = 4000) => {
    setToaster({ show: true, message, type });
    const timer = setTimeout(() => setToaster(prev => ({ ...prev, show: false })), duration);
    return () => clearTimeout(timer); // Cleanup timer on unmount or new call
  }, []);

  // Font class based on script
  const getFontClass = useCallback((script) => {
    if (script === 'te') return 'font-tiro-telugu';
    if (script === 'hi' || script === 'en') return 'font-tiro-devanagari-sanskrit';
    return 'font-mono';
  }, []);


  // Audio Event Listeners
  useEffect(() => {
    const audio = audioRef.current;
    if (!audio || !hasActualInteractiveSegments) return;

    const handlePlay = () => setIsPlaying(true);
    const handlePause = () => setIsPlaying(false);
    const handleLoadedMetadata = () => setAudioDuration(audio.duration);
    const handleTimeUpdate = () => {
      setAudioCurrentTime(audio.currentTime);
      // Auto-pause at the end of a segment if one is playing
      if (currentPlayingSegmentId && currentInteractiveData && selectedAudioSourceId) {
        const segment = currentInteractiveData.segments.find(s => s.id === currentPlayingSegmentId);
        const timestampData = segment?.timestamps[selectedAudioSourceId];
        if (timestampData && timestampData.end > 0 && audio.currentTime >= timestampData.end) {
          audio.pause();
          setCurrentPlayingSegmentId(null); // Deselect segment
        }
      }
    };
    const handleEnded = () => {
      setIsPlaying(false);
      setCurrentPlayingSegmentId(null);
      if (audioRef.current) setAudioCurrentTime(audioRef.current.duration); // Show full duration
    };

    audio.addEventListener('play', handlePlay);
    audio.addEventListener('pause', handlePause);
    audio.addEventListener('loadedmetadata', handleLoadedMetadata);
    audio.addEventListener('timeupdate', handleTimeUpdate);
    audio.addEventListener('ended', handleEnded);

    return () => {
      audio.removeEventListener('play', handlePlay);
      audio.removeEventListener('pause', handlePause);
      audio.removeEventListener('loadedmetadata', handleLoadedMetadata);
      audio.removeEventListener('timeupdate', handleTimeUpdate);
      audio.removeEventListener('ended', handleEnded);
    };
  }, [audioRef, hasActualInteractiveSegments, currentPlayingSegmentId, currentInteractiveData, selectedAudioSourceId]);

  // Handle Audio Source Change
  useEffect(() => {
    const audio = audioRef.current;
    if (hasActualInteractiveSegments && currentInteractiveData && selectedAudioSourceId && audio) {
      const newSrc = currentInteractiveData.audioSources.find(s => s.id === selectedAudioSourceId)?.src;
      audio.pause();
      setCurrentPlayingSegmentId(null);
      setIsPlaying(false);
      setAudioCurrentTime(0);
      setAudioDuration(0); // Reset duration until new metadata loads

      if (newSrc) {
        const isPlaceholder = newSrc.includes('placeholder_');
        // Ensure full URL for comparison and setting src
        const newFullSrc = (typeof window !== 'undefined' && window.location) ? (new URL(newSrc, window.location.origin)).href : newSrc;

        if (audio.currentSrc !== newFullSrc) { // Only change src if it's different
          audio.src = newSrc;
          if (!isPlaceholder) audio.load(); // Load new audio if not placeholder
          else setAudioDuration(0); // For placeholders, duration is 0
        } else if (isPlaceholder) { // If src is same but it's a placeholder
          setAudioDuration(0);
        }
      } else { // No source selected or available
        if (audio.src) audio.removeAttribute('src'); // Remove src if it exists
        if (audio.load) audio.load(); // Reset audio element
        setAudioDuration(0);
      }
    } else if (audio && audio.src && !hasActualInteractiveSegments) { // If no interactive segments, clear audio
      audio.pause();
      audio.removeAttribute('src');
      if (audio.load) audio.load();
      setIsPlaying(false);
      setCurrentPlayingSegmentId(null);
      setAudioCurrentTime(0);
      setAudioDuration(0);
    }
  }, [hasActualInteractiveSegments, selectedAudioSourceId, currentInteractiveData, audioRef]);


  // Handle Playback Speed Change
  useEffect(() => {
    if (audioRef.current) audioRef.current.playbackRate = playbackSpeed;
  }, [playbackSpeed, audioRef]);

  // Handle Segment Click
  const handleSegmentClick = (segment) => {
    const audio = audioRef.current;
    if (!hasActualInteractiveSegments || !currentInteractiveData || !selectedAudioSourceId || !audio) return;

    const isPlaceholder = audio.currentSrc && audio.currentSrc.includes('placeholder_');
    const timestamps = segment.timestamps[selectedAudioSourceId];

    if (currentPlayingSegmentId === segment.id) { // If clicking the currently playing segment
      handlePlayPause(); // Toggle play/pause
      return;
    }

    setCurrentPlayingSegmentId(segment.id);
    if (timestamps && typeof timestamps.start === 'number' && typeof timestamps.end === 'number' && timestamps.end > timestamps.start && timestamps.end > 0) {
      setAudioCurrentTime(timestamps.start); // Set current time for visual feedback
      if (isPlaceholder) {
        displayToaster("Placeholder audio. Interactive playback disabled.", "info");
        setIsPlaying(false); // Ensure it's not marked as playing
        return;
      }
      if (audio.readyState >= audio.HAVE_METADATA) { // Check if audio is ready
        audio.currentTime = timestamps.start;
        audio.play().catch(e => {
          displayToaster(`Error playing audio: ${e.message}`, "error");
          setIsPlaying(false);
        });
      } else {
        displayToaster("Audio is still loading. Please wait.", "info");
        if(audio.src && !isPlaceholder) audio.load(); // Trigger load if not already loading
      }
    } else {
      displayToaster("Audio timing data unavailable for this segment/teacher.", "warning");
    }
  };

  // Handle Play/Pause Button
  const handlePlayPause = () => {
    const audio = audioRef.current;
    if (!audio) return;

    const isPlaceholder = audio.currentSrc && audio.currentSrc.includes('placeholder_');
    if (isPlaceholder) {
        displayToaster("Placeholder audio. Playback disabled.", "info");
        setIsPlaying(false);
        return;
    }

    if (!audio.src || audio.readyState < audio.HAVE_METADATA) { // Audio not loaded or no src
      displayToaster("Audio not ready or no source selected.", "warning");
      return;
    }

    if (isPlaying) {
      audio.pause();
    } else {
      // If at the end of audio, restart from beginning (or current segment start if applicable)
      if (!currentPlayingSegmentId && audioDuration > 0 && audioCurrentTime >= audioDuration - 0.1) {
        audio.currentTime = 0;
        setAudioCurrentTime(0);
      }
      audio.play().catch(e => {
        displayToaster(`Error playing audio: ${e.message}`, "error");
        setIsPlaying(false);
      });
    }
  };

  // Handle Stop Button
  const handleStop = () => {
    const audio = audioRef.current;
    if (!audio) return;
    audio.pause();
    audio.currentTime = 0;
    setCurrentPlayingSegmentId(null);
    setAudioCurrentTime(0);
    setIsPlaying(false);
  };

  // Handle Timeline Scrubbing
  const handleTimelineChange = (event) => {
    const audio = audioRef.current;
    if (audio && audioDuration > 0 && audio.seekable && audio.seekable.length > 0) { // Check if audio is seekable
      const newTime = parseFloat(event.target.value);
      audio.currentTime = newTime;
      setAudioCurrentTime(newTime);
    }
  };

  // Effect to reset state when selectedChapter changes
  useEffect(() => {
    const chapterKey = `${selectedChapter.trackId}_${selectedChapter.id}`;
    const newInteractiveData = INTERACTIVE_CHAPTER_DATA[chapterKey];
    const newHasActualInteractiveSegments = newInteractiveData?.segments?.length > 0 && !newInteractiveData.segments[0].id.startsWith('ph_seg_');

    // Reset relevant states
    setSelectedAudioSourceId(newInteractiveData?.audioSources[0]?.id || null);
    setCurrentPlayingSegmentId(null);
    setHoveredSegmentId(null);
    setIsPlaying(false);
    setPlaybackSpeed(1);
    setAudioCurrentTime(0);
    setAudioDuration(0);

    const audio = audioRef.current;
    if (audio) {
      audio.pause();
      audio.currentTime = 0;
      if (!newHasActualInteractiveSegments && audio.currentSrc) {
        audio.removeAttribute('src'); // Remove src if no interactive segments
        if(audio.load) audio.load(); // Reset audio element state
      } else if (newHasActualInteractiveSegments && newInteractiveData) {
        const initialAudioSrc = newInteractiveData.audioSources[0]?.src;
        const newFullSrc = initialAudioSrc ? (new URL(initialAudioSrc, window.location.origin)).href : null;

        if (initialAudioSrc && audio.currentSrc !== newFullSrc) {
          audio.src = initialAudioSrc;
          if (!initialAudioSrc.includes('placeholder_')) audio.load();
          else setAudioDuration(0); // Placeholder
        } else if (!initialAudioSrc && audio.currentSrc) { // No initial source, clear existing
            audio.removeAttribute('src');
            if(audio.load) audio.load();
            setAudioDuration(0);
        } else if (initialAudioSrc && initialAudioSrc.includes('placeholder_')) {
            setAudioDuration(0); // Already placeholder or src is same
        }
      }
    }
  }, [selectedChapter, audioRef]); // Rerun when selectedChapter or audioRef changes

  return (
    <div className="p-4 sm:p-6 max-w-4xl mx-auto">
      {/* Toaster Notifications */}
      {toaster.show && (
        <div className={`fixed top-5 right-5 p-4 rounded-md shadow-lg text-white z-50 flex items-center ${
          toaster.type === 'error' ? 'bg-red-500' : toaster.type === 'info' ? 'bg-blue-500' : 'bg-yellow-500'
        }`}>
          <Info size={20} className="mr-2"/>
          {toaster.message}
        </div>
      )}

      {/* Navigation and Header */}
      <div className="mb-6">
        <button onClick={() => setCurrentView('studentTrackView')} className="flex items-center gap-2 text-blue-600 hover:text-blue-800 mb-4">
          <ChevronLeft size={20} />Back to Track {selectedTrack.id}
        </button>
        <div className="flex flex-col sm:flex-row justify-between items-start gap-4 mb-4 p-5 bg-white rounded-xl shadow-lg border">
          <div className="flex-grow">
            <p className="text-xs sm:text-sm text-gray-500 mb-1">Track {selectedTrack.id} • Chapter {selectedChapter.id}</p>
            <h1 className="text-2xl sm:text-3xl font-bold text-gray-800 break-words">{currentChapterFullTextData?.title || selectedChapter.title}</h1>
          </div>
          <div className={`mt-2 sm:mt-0 px-3 py-1.5 rounded-full text-sm font-medium shadow-sm ${getLevelColor(chapterProficiencyLevel)} flex-shrink-0`}>
            Proficiency: {getLevelLabel(chapterProficiencyLevel)}
          </div>

      <MediaControls
        audioSources={currentInteractiveData?.audioSources || []}
        selectedAudioSourceId={selectedAudioSourceId}
        onAudioSourceChange={setSelectedAudioSourceId}
        currentPlaybackSpeed={playbackSpeed}
        onPlaybackSpeedChange={setPlaybackSpeed}

const StudentProficiencyManagementView = ({ student, tracks, studentProficiencies, updateStudentProficiency, setCurrentView, setSelectedStudentForManagement, getLevelColor, getLevelLabel }) => {
  if (!student) return null;

  const [currentProficiencies, setCurrentProficiencies] = useState(studentProficiencies[student.id] || {});
  const [expandedTrackId, setExpandedTrackId] = useState(null); // Track which track is expanded

  const handleProficiencyChange = (chapterKey, newLevel) => {
    setCurrentProficiencies(prev => ({ ...prev, [chapterKey]: parseInt(newLevel, 10) }));
  };

  const handleTrackLevelChange = (trackId, newLevelStr) => {
    const level = parseInt(newLevelStr, 10);
    if (isNaN(level)) return; // Do nothing if not a number

    const updatedProficienciesForTrack = { ...currentProficiencies };
    const targetTrack = tracks.find(t => t.id === trackId);
    if (targetTrack) {
      targetTrack.chapters.forEach(chapter => {
        updatedProficienciesForTrack[`${trackId}_${chapter.id}`] = level;
      });
    }
    setCurrentProficiencies(updatedProficienciesForTrack);
  };


  const handleSaveChanges = () => {
    updateStudentProficiency(student.id, currentProficiencies);
    alert(`Proficiency for ${student.name} saved!`); // Replace with a proper notification system later
  };

  // Determine if a track's proficiency can be updated (previous track must be at least Level 2)
  const canUpdateTrackProficiency = (trackId) => {
    if (trackId === 1) return true; // First track is always updatable
    const prevTrack = tracks.find(t => t.id === trackId - 1);
    if (!prevTrack) return true; // Should not happen if tracks are sequential
    return prevTrack.chapters.every(ch => (currentProficiencies[`${prevTrack.id}_${ch.id}`] ?? 0) >= 2);
  };

  const toggleTrackExpansion = (trackId) => {
    setExpandedTrackId(prev => prev === trackId ? null : trackId);
  };

  return (
    <div className="p-4 sm:p-6 max-w-5xl mx-auto">
      <div className="flex flex-col sm:flex-row justify-between sm:items-center gap-4 mb-6">
        <div>
          <button
            onClick={() => { setCurrentView('unifiedDashboard'); setSelectedStudentForManagement(null); }}
            className="flex items-center gap-2 text-blue-600 hover:text-blue-800 mb-2 text-sm"
          >
            <ChevronLeft size={18}/>Back to Instructor Panel
          </button>
          <h1 className="text-2xl sm:text-3xl font-bold text-gray-800">Manage Proficiency: {student.name}</h1>
        </div>
        <button
          onClick={handleSaveChanges}
          className="px-6 py-2 bg-green-600 text-white rounded-lg hover:bg-green-700 font-medium flex items-center gap-2 self-start sm:self-center"
        >
          <CheckCircle size={18}/>Save Changes
        </button>
      </div>

      {tracks.map(track => {
        const isTrackProficiencyUpdatable = canUpdateTrackProficiency(track.id);
        const isExpanded = expandedTrackId === track.id;
        return(
        <div key={track.id} className="mb-6 bg-white rounded-xl shadow-lg border border-gray-200 overflow-hidden">
          <div
            className="flex justify-between items-center p-4 cursor-pointer hover:bg-gray-50"
            onClick={() => toggleTrackExpansion(track.id)}
          >
            <h2 className="text-xl font-semibold text-gray-700">{`Track ${track.id}: ${track.title}`}</h2>
            <div className="flex items-center gap-3">
              {!isTrackProficiencyUpdatable && (
                <span className="text-xs font-medium text-red-600 bg-red-100 px-2 py-1 rounded-full flex items-center gap-1">
                  <Lock size={12}/>Track {track.id-1} Incomplete
                </span>
              )}
              <div className="p-2 rounded-full hover:bg-gray-200">
                {isExpanded ? <ChevronUp size={20} className="text-gray-600" /> : <ChevronDown size={20} className="text-gray-600" />}
              </div>
            </div>
          </div>

          {isExpanded && (
            <div className="p-4 border-t border-gray-200">
              {/* Set all chapters in track to a level */}
              <div className="mb-4 flex items-center gap-3 border-b pb-3">
                <label htmlFor={`track-${track.id}-level`} className="text-sm font-medium text-gray-600 whitespace-nowrap">Set all in track to:</label>
                <select

                  onChange={(e) => handleTrackLevelChange(track.id, e.target.value)}

                {track.chapters.map(chapter => {

                          <select
                            value={currentLevel}

                        )}
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          )}
        </div>
      )})}
    </div>
  );
};

const UserFormModal = ({ user, roles, onSave, onCancel, allAppRoles }) => {
  const [formData, setFormData] = useState({
    email: '',
    displayName: '',
    roles: [],
    ...user // Spread existing user data if editing
  });

  useEffect(() => {
    if (user) {
      setFormData({ email: user.email || '', displayName: user.displayName || '', roles: user.roles || [] });
    } else { // For new user
      setFormData({ email: '', displayName: '', roles: ['student'] }); // Default new user to student
    }
  }, [user]);

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  const handleRoleChange = (role) => {
    let newRoles = [...formData.roles];
    if (newRoles.includes(role)) {
      newRoles = newRoles.filter(r => r !== role);
      // If unchecking instructor, student role remains (can be manually unchecked unless other logic prevents)
    } else {
      newRoles.push(role);
      if (role === 'instructor' && !newRoles.includes('student')) {
        newRoles.push('student'); // Auto-select student if instructor is selected
      }
    }
    setFormData(prev => ({ ...prev, roles: newRoles }));
  };
  
  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.email && !user) { // Only require email for new user creation
        alert("Email is required for a new user."); // Replace with better UI feedback
        return;
    }
    if (formData.roles.length === 0) {
        alert("User must have at least one role."); // Replace with better UI feedback
        return;
    }
    // Ensure student role is present if instructor role is, even if somehow deselected through other means (defensive)
    if (formData.roles.includes('instructor') && !formData.roles.includes('student')) {
        formData.roles.push('student');
    }
    onSave({ ...user, ...formData });
  };

  const isNewUser = !user || !user.id;

  return (
    <div className="fixed inset-0 bg-gray-600 bg-opacity-50 overflow-y-auto h-full w-full z-50 flex justify-center items-center p-4">
      <div className="bg-white p-6 sm:p-8 rounded-lg shadow-xl w-full max-w-lg max-h-full overflow-y-auto">
        <h2 className="text-2xl font-semibold mb-6 text-gray-800">{isNewUser ? 'Invite New User' : 'Edit User'}</h2>
        <form onSubmit={handleSubmit}>
          <div className="mb-4">
            <label htmlFor="email" className="block text-sm font-medium text-gray-700 mb-1">Email Address</label>
            <input type="email" name="email" id="email" value={formData.email} onChange={handleInputChange} 
                   className="w-full p-2 border border-gray-300 rounded-md focus:ring-indigo-500 focus:border-indigo-500 text-sm" 
                   required disabled={!isNewUser} />
            {!isNewUser && <p className="text-xs text-gray-500 mt-1">Email cannot be changed after account creation.</p>}
          </div>
          <div className="mb-4">
            <label htmlFor="displayName" className="block text-sm font-medium text-gray-700 mb-1">Display Name</label>
            <input type="text" name="displayName" id="displayName" value={formData.displayName} onChange={handleInputChange} 
                   className="w-full p-2 border border-gray-300 rounded-md focus:ring-indigo-500 focus:border-indigo-500 text-sm" />
          </div>
          <div className="mb-6">
            <label className="block text-sm font-medium text-gray-700 mb-2">Roles</label>

                <td className="px-4 py-3 whitespace-nowrap">
                  <div className="text-sm font-medium text-gray-900">{user.displayName || '(No Name)'}</div>
                  <div className="text-xs text-gray-500">{user.email}</div>
                </td>
                <td className="px-4 py-3 whitespace-nowrap">
                  <div className="flex flex-wrap gap-1">
                    {user.roles.map(role => (
                      <span key={role} className={`px-2 py-0.5 text-xs font-semibold rounded-full ${roleColors[role] || 'bg-gray-100 text-gray-600'}`}>
                        {role.replace('_', ' ')}
                      </span>
                    ))}
                  </div>
                </td>
                <td className="px-4 py-3 whitespace-nowrap">
                  <span className={`px-2.5 py-1 inline-flex text-xs leading-5 font-semibold rounded-full ${getStatusColor(user.status)}`}>
                    {user.status.replace('_', ' ')}
                  </span>
                </td>
                <td className="px-4 py-3 whitespace-nowrap text-xs text-gray-500">
                  {user.status === 'pending_invitation' && user.inviteSentAt ? 
                    `Invited ${calculateDaysSince(user.inviteSentAt)} days ago` : 
                    (user.lastLogin ? `Logged in: ${new Date(user.lastLogin).toLocaleDateString()}` : 'N/A')
                  }
                </td>
                <td className="px-4 py-3 whitespace-nowrap text-sm font-medium">
                  <div className="flex items-center space-x-2">
                    <button onClick={() => handleEditUser(user)} className="text-indigo-600 hover:text-indigo-800 p-1" title="Edit User"><Edit3 size={16}/></button>
                    {user.status === 'pending_invitation' && (
                      <button onClick={() => handleResendInvite(user.id)} className="text-blue-600 hover:text-blue-800 p-1" title="Resend Invite"><MailWarning size={16}/></button>
                    )}
                    {user.status !== 'pending_invitation' && ( // Toggle status only if not pending
                        <button onClick={() => handleToggleStatus(user.id)} 
                                className={`${user.status === 'active' ? 'text-red-600 hover:text-red-800' : 'text-green-600 hover:text-green-800'} p-1`}
                                title={user.status === 'active' ? 'Disable Account' : 'Enable Account'}>
                          {user.status === 'active' ? <ShieldOff size={16}/> : <Shield size={16}/>}
                        </button>
                    )}
                  </div>
                </td>
              </tr>
            ))}
             {filteredUsers.length === 0 && (
                <tr><td colSpan="5" className="text-center py-8 text-gray-500">No users match your criteria.</td></tr>
            )}
          </tbody>
        </table>
      </div>
      {showUserForm && <UserFormModal user={editingUser} onSave={handleSaveUser} onCancel={() => {setShowUserForm(false); setEditingUser(null);}} allAppRoles={ALL_ROLES} />}
    </div>
  );
};

// Unified Dashboard for switching between Student, Instructor, Admin and Content Manager views
const UnifiedDashboardView = ({ 
  tracks, 
  students: instructorStudents, // Renamed to avoid conflict if a 'students' prop is used for something else
  setCurrentView, 
  setSelectedTrack, 
  setSelectedStudentForManagement, 
  studentProficiency, 
  getLevelColor, 
  getLevelLabel, 
  isTrackNotStarted, 
  isTrackComplete, 
  isTrackReadyForTest,
  currentAppUserRoles // Roles of the currently logged-in user
}) => {
  const [activeTab, setActiveTab] = useState(''); 

  const availableTabs = useMemo(() => {
    const tabs = [];
    if (currentAppUserRoles.includes('student')) tabs.push({ id: 'learning', label: 'My Learning', icon: BookOpen });
    if (currentAppUserRoles.includes('instructor')) tabs.push({ id: 'instructor', label: 'Instructor Panel', icon: Users });
    if (currentAppUserRoles.includes('content_manager')) tabs.push({ id: 'content_manager', label: 'Manage Content', icon: Briefcase });
    if (currentAppUserRoles.includes('admin')) tabs.push({ id: 'admin', label: 'Administration', icon: Settings });
    return tabs;
  }, [currentAppUserRoles]);

 // Effect to set the active tab based on available roles
 useEffect(() => {
    if (availableTabs.length > 0) {
      // If current activeTab is not valid or not set, default to the first available tab
      if (!activeTab || !availableTabs.some(tab => tab.id === activeTab)) {
        setActiveTab(availableTabs[0].id);
      }
    } else {
      setActiveTab(''); // No tabs available
    }
  }, [availableTabs, activeTab]);


  return (
    <div className="p-4 sm:p-6 max-w-full mx-auto"> 
        <div className="mb-8 text-center">
            <div className="text-5xl sm:text-6xl mb-4 inline-block"> 🕉️ </div>
            <h1 className="text-3xl sm:text-4xl font-bold text-gray-800 mb-2">Vedic Learning System</h1>
            <p className="text-md sm:text-lg text-gray-600">Student & Instructor Portal</p>
        </div>
      
        {availableTabs.length > 0 ? (
            <div className="mb-6 flex flex-wrap border-b border-gray-300">
                {availableTabs.map(tabInfo => (
                <button 
                    key={tabInfo.id}
                    onClick={() => setActiveTab(tabInfo.id)} 
                    className={`flex items-center justify-center gap-2 px-4 sm:px-6 py-3 font-medium text-center flex-grow sm:flex-grow-0 basis-1/2 sm:basis-auto ${activeTab === tabInfo.id ? 'border-b-2 border-indigo-600 text-indigo-600' : 'text-gray-500 hover:text-gray-700'}`}
                >
                    <tabInfo.icon size={18} />{tabInfo.label}
                </button>
                ))}
            </div>
        ) : (
            <div className="text-center py-10">
                <Info size={48} className="mx-auto text-gray-400 mb-4" />
                <p className="text-xl text-gray-600">No functionalities available for your current role.</p>
                 <p className="text-sm text-gray-500 mt-2">Please contact an administrator if you believe this is an error.</p>
            </div>
        )}

      {activeTab === 'learning' && currentAppUserRoles.includes('student') && ( <StudentDashboardView tracks={tracks} setCurrentView={setCurrentView} setSelectedTrack={setSelectedTrack} studentProficiency={studentProficiency} getLevelColor={getLevelColor} getLevelLabel={getLevelLabel} isTrackNotStarted={isTrackNotStarted} isTrackComplete={isTrackComplete} isTrackReadyForTest={isTrackReadyForTest} /> )}
      {activeTab === 'instructor' && currentAppUserRoles.includes('instructor') && ( <InstructorStudentsListView students={instructorStudents} setCurrentView={setCurrentView} setSelectedStudentForManagement={setSelectedStudentForManagement}/> )}
        {activeTab === 'admin' && currentAppUserRoles.includes('admin') && ( <AdminView /> )}
        {activeTab === 'content_manager' && currentAppUserRoles.includes('content_manager') && ( <ContentManagerPlaceholderView /> )}
    </div>
  );
};

const App = () => {
  // View and Selection State
  const [currentView, setCurrentView] = useState('unifiedDashboard'); // Default to unified dashboard

  // Default to all roles for initial full view.

  // Student's own proficiency (for "My Learning" tab)
  // Initialize with default levels from TRACKS_DATA
  const [currentUserProficiency, setCurrentUserProficiency] = useState(
    TRACKS_DATA.reduce((acc, track) => {
      track.chapters.forEach(ch => {
        acc[`${track.id}_${ch.id}`] = ch.level; // Use initial level from data
      });
      return acc;
    }, {})
  );

  // Data for Instructor Panel
  const [studentsData] = useState(MOCK_STUDENTS_DATA_INIT); // Static for now
  const [studentProficiencies, setStudentProficiencies] = useState(
    MOCK_STUDENTS_DATA_INIT.reduce((acc, student) => {
      acc[student.id] = { ...(student.chapterProficiency || {}) };
      // Ensure all chapters have a proficiency entry, defaulting to 0
      TRACKS_DATA.forEach(track => {
        track.chapters.forEach(chapter => {
          const chapterKey = `${track.id}_${chapter.id}`;
          if (acc[student.id][chapterKey] === undefined) {
            acc[student.id][chapterKey] = 0;
          }
        });
      });
      return acc;
    }, {})
  );
  const [selectedStudentForManagement, setSelectedStudentForManagement] = useState(null);

  // Audio Ref
  const audioRef = useRef(null);

  // Memoized helper functions for performance
  const memoizedGetLevelColor = useCallback(getLevelColor, []);
  const memoizedGetLevelLabel = useCallback(getLevelLabel, []);

  const memoizedIsStudentTrackComplete = useCallback((trackWithLevels) => {
    if (!trackWithLevels || !trackWithLevels.chapters) return false;
    return trackWithLevels.chapters.every(chapter => chapter.level === 4);
  }, []);

  const memoizedIsStudentTrackReadyForTest = useCallback((trackWithLevels) => {
    if (!trackWithLevels || !trackWithLevels.chapters) return false;
    return trackWithLevels.chapters.every(chapter => chapter.level >= 3) && !memoizedIsStudentTrackComplete(trackWithLevels);
  }, [memoizedIsStudentTrackComplete]); 

  const memoizedIsStudentTrackNotStarted = useCallback((trackWithLevels) => {
    if (!trackWithLevels || !trackWithLevels.chapters) return false;
    return trackWithLevels.chapters.every(chapter => chapter.level === 0);
  }, []);

  // Function to update proficiency for a student (used by instructor panel)
  const updateStudentProficiency = (studentId, newProficiencies) => {
    setStudentProficiencies(prev => ({ ...prev, [studentId]: newProficiencies }));
  };
  
  // This function was for testing and is now removed as per user request.
  // const cycleUserRoles = () => { ... };


  return (
    <div className="min-h-screen bg-gradient-to-br from-orange-50 via-amber-50 to-yellow-50 font-sans text-slate-800">
      {/* The "Cycle Roles" button and its container div have been removed */}

      {currentView === 'unifiedDashboard' && (
        <UnifiedDashboardView 
            tracks={TRACKS_DATA} 
            students={studentsData} 
            setCurrentView={setCurrentView} 
            setSelectedTrack={setSelectedTrack} 
            setSelectedStudentForManagement={setSelectedStudentForManagement} 
            studentProficiency={currentUserProficiency} // This is for the "My Learning" tab
            getLevelColor={memoizedGetLevelColor} 
            getLevelLabel={memoizedGetLevelLabel} 
            isTrackNotStarted={memoizedIsStudentTrackNotStarted} 
            isTrackComplete={memoizedIsStudentTrackComplete} 
            isTrackReadyForTest={memoizedIsStudentTrackReadyForTest}
            currentAppUserRoles={currentAppUserRoles}
        /> 
      )}
      {currentView === 'studentTrackView' && selectedTrack && ( <StudentTrackView selectedTrack={selectedTrack} setCurrentView={setCurrentView} setSelectedChapter={setSelectedChapter} studentProficiency={currentUserProficiency} getLevelColor={memoizedGetLevelColor} getLevelLabel={memoizedGetLevelLabel} isTrackComplete={memoizedIsStudentTrackComplete} isTrackReadyForTest={memoizedIsStudentTrackReadyForTest} /> )}
      {currentView === 'studentChapterView' && selectedChapter && selectedTrack && ( <StudentChapterView selectedChapter={selectedChapter} selectedTrack={selectedTrack} setCurrentView={setCurrentView} audioRef={audioRef} studentProficiency={currentUserProficiency} getLevelColor={memoizedGetLevelColor} getLevelLabel={memoizedGetLevelLabel}/> )}
      {currentView === 'instructorStudentProficiencyManagementView' && selectedStudentForManagement && ( <StudentProficiencyManagementView student={selectedStudentForManagement} tracks={TRACKS_DATA} studentProficiencies={studentProficiencies} updateStudentProficiency={updateStudentProficiency} setCurrentView={setCurrentView} setSelectedStudentForManagement={setSelectedStudentForManagement} getLevelColor={memoizedGetLevelColor} getLevelLabel={memoizedGetLevelLabel}/> )}
      <audio ref={audioRef} className="hidden" /> {/* Hidden audio element controlled by JS */}
    </div>
  );
};
export default App;
